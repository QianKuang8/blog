Initialize the same collection as that template: a bare .Paginator here would cache a different list. */ -}} {{- if or (eq .Kind "term") (and .IsSection (eq .Section "posts")) -}} {{- $pager := .Paginate (union .RegularPages .Sections) -}} {{- if gt $pager.PageNumber 1 -}} {{- $url = $pager.URL | absURL -}} {{- end -}} {{- end -}} {{- with .Params.canonicalURL -}} {{- $url = trim . " " -}} {{- end -}} {{- return $url -}}
